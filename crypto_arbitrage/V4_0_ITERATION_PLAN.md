```python
class KnowledgeCycleManager:
    def __init__(self, recorder, assistant):
        self.recorder = recorder
        self.assistant = assistant

    async def cycle(self, decision_context):
        """执行知识循环"""
        # 1. 获取决策建议
        advice = await self.assistant.get_advice(decision_context)

        # 2. 执行决策
        result = await self._execute(decision_context, advice)

        # 3. 记录知识
        await self.recorder.record(decision_context, advice, result)

        # 4. 分析和学习
        await self._learn_and_optimize()

        return result
```
